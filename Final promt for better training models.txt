        with conn.cursor() as cur:
            cur.execute("""
                SELECT column_name 
                FROM information_schema.columns 
                WHERE table_schema = 'public' 
                AND table_name = 'training_logs' 
                AND column_name = 'model_name';
            """)
            if not cur.fetchone():

                        accuracy FLOAT,
                        f1_score FLOAT,
                        class_accuracies JSON,

                """)

            else:
                print("Table 'training_logs' exists with 'model_name' column.")
        conn.commit()
        return conn
    except psycopg2.Error as e:
        print(f"Failed to verify or create table: {e}")
        conn.rollback()
        return None

class CocoDataset(Dataset):
    def __init__(self, coco, data_dir, transform=None, phase='train', target_class=None):
        self.coco = coco
        self.data_dir = data_dir
        self.transform = transform
        self.phase = phase
        self.target_class = target_class
        self.ppe_map = {name: idx for idx, name in enumerate(PPE_ITEMS)}
        self.img_ids = coco.getImgIds()
        self.valid_img_ids = []
        self.img_to_anns = {}
        self.skipped_ids = []
        self._filter_valid_images()
        self.class_imgs = {cls: [] for cls in PPE_ITEMS} if not target_class else {target_class: []}
        self.update_class_distribution()
        print(f"Dataset Phase: {phase}, Target Class: {target_class}, Images: {len(self.valid_img_ids)} (Skipped {len(self.skipped_ids)})")

    def _filter_valid_images(self):
        print(f"Filtering {len(self.img_ids)} images for validity...")
        excluded_filename = "US_Navy_110416-N-VL413-001_Japanese_civilian_shipyard_workers_aboard_the_nuclear-powered_aircraft_carrier_USS_George_Washington_-28CVN_73-29_move_a_mot_jpg.rf.d01541534a343898af2527637bbb4d90.jpg"
        for img_id in self.img_ids:
            img_info = self.coco.loadImgs([img_id])[0]
            if img_info['file_name'] == excluded_filename:
                print(f"Excluding image ID {img_id}: {img_info['file_name']}")
                self.skipped_ids.append(img_id)
                continue
            img_path = os.path.join(self.data_dir, img_info['file_name'])
            if not os.path.exists(img_path):
                base_name = img_info['file_name'].split('.rf.')[0].rsplit('.', 1)[0]
                found = False
                for ext in ['.jpg', '.jpeg', '.png']:
                    test_path = os.path.join(self.data_dir, base_name + ext)
                    if os.path.exists(test_path):
                        img_path = test_path
                        found = True
                        break
                if not found:
                    print(f"Skipping invalid image ID {img_id}: {img_info['file_name']}")
                    self.skipped_ids.append(img_id)
                    continue
            self.valid_img_ids.append(img_id)
            self.img_to_anns[img_id] = self.coco.loadAnns(self.coco.getAnnIds(imgIds=img_id))
        print(f"Valid images: {len(self.valid_img_ids)}, Skipped: {len(self.skipped_ids)}")

    def update_class_distribution(self):
        if not self.target_class:
            for cls in self.class_imgs:
                self.class_imgs[cls] = []
            for img_id in self.valid_img_ids:
                anns = self.img_to_anns[img_id]
                labels = set()
                for ann in anns:
                    category_name = self.coco.loadCats(ann['category_id'])[0]['name']
                    if category_name in self.ppe_map:
                        labels.add(category_name)
                    elif category_name in ["NO-Hardhat", "NO-Safety Vest", "NO-Gloves", "NO-Goggles", "NO-Mask"]:
                        labels.add(category_name)
                    elif category_name == "Fall-Detected":
                        labels.add("Fall-Detected")
                    elif category_name == "Person":
                        labels.add("Person")
                if not labels:
                    labels.add("NO-Hardhat")
                for label in labels:
                    if label in self.ppe_map:
                        self.class_imgs[label].append(img_id)
        else:
            target_class_id = next((cat_id for cat_id, cat in self.coco.cats.items() if cat['name'] == self.target_class), None)
            if target_class_id:
                for img_id in self.valid_img_ids:
                    anns = self.img_to_anns[img_id]
                    if any(ann['category_id'] == target_class_id for ann in anns):
                        self.class_imgs[self.target_class].append(img_id)
        class_counts = {cls: len(imgs) for cls, imgs in self.class_imgs.items()}
        print(f"Class Distribution ({self.phase}): {class_counts}")

    def __len__(self):
        return len(self.valid_img_ids)

class BalancedSampler(torch.utils.data.Sampler):
    def __init__(self, dataset, oversample_factor=2.0):
        self.dataset = dataset
        self.oversample_factor = oversample_factor
        self.class_imgs = {cls: [i for i in range(len(self.dataset.valid_img_ids)) if any(
            self.dataset.coco.loadCats(ann['category_id'])[0]['name'] in self.dataset.ppe_map
            for ann in self.dataset.img_to_anns[self.dataset.valid_img_ids[i]]
        )] for cls in dataset.class_imgs.keys() if not dataset.target_class}
        self.weights = self._calculate_weights()
        print(f"Sampler class distribution: { {cls: len(imgs) for cls, imgs in self.class_imgs.items()} }")

    def _calculate_weights(self):
        class_counts = {cls: len(indices) for cls, indices in self.class_imgs.items()}
        max_count = max(class_counts.values()) if class_counts else 1
        class_weights = {cls: max(max_count / (count + 1e-6), 1e-6) for cls, count in class_counts.items()}
        img_weights = torch.zeros(len(self.dataset.valid_img_ids))
        for cls, indices in self.class_imgs.items():
            for idx in indices:
                img_weights[idx] = class_weights[cls]
        img_weights = torch.where(img_weights == 0, torch.tensor(1e-6, dtype=img_weights.dtype), img_weights)
        return img_weights

    def __iter__(self):
        weights = self.weights.double()
        num_samples = int(len(self.dataset.valid_img_ids) * self.oversample_factor)
        indices = torch.multinomial(weights, num_samples, replacement=True)
        return iter(indices.tolist())

    def __len__(self):
        return int(len(self.dataset.valid_img_ids) * self.oversample_factor)

class EnhancedCustomCNN(nn.Module):
    def __init__(self, num_classes=len(PPE_ITEMS)):
        super(EnhancedCustomCNN, self).__init__()
        self.features = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 32, kernel_size=3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.MaxPool2d(2, 2), nn.Dropout2d(0.2),
            nn.Conv2d(32, 64, kernel_size=3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 64, kernel_size=3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.MaxPool2d(2, 2), nn.Dropout2d(0.2),
            nn.Conv2d(64, 128, kernel_size=3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
            nn.Conv2d(128, 128, kernel_size=3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
            nn.MaxPool2d(2, 2), nn.Dropout2d(0.2),
            nn.Conv2d(128, 256, kernel_size=3, padding=1), nn.BatchNorm2d(256), nn.ReLU(),
            nn.Conv2d(256, 256, kernel_size=3, padding=1), nn.BatchNorm2d(256), nn.ReLU(),
            nn.MaxPool2d(2, 2), nn.Dropout2d(0.2)
        )
        self.feature_size = 256 * (224 // 16) * (224 // 16)
        self.classifier = nn.Sequential(
            nn.Linear(self.feature_size, 512), nn.ReLU(), nn.Dropout(0.5),
            nn.Linear(512, 256), nn.ReLU(), nn.Dropout(0.5),
            nn.Linear(256, num_classes if num_classes > 1 else 1)
        )

    def forward(self, x):
        x = self.features(x)
        x = x.view(x.size(0), -1)
        logits = self.classifier(x)
        return logits, x

class EnsembleModel:
    def __init__(self, models, weights=None):
        self.models = models
        self.weights = weights if weights else [1/len(models)] * len(models)

    def predict(self, x):
        outputs = []
        features_list = []
        for i, model in enumerate(self.models):
            model.eval()
            with torch.no_grad():
                logits, features = model(x)
                outputs.append(self.weights[i] * torch.sigmoid(logits))
                features_list.append(features)
        ensemble_output = torch.zeros_like(outputs[0])
        for output in outputs:
            ensemble_output += output
        return ensemble_output, features_list[0]

def train_model(model, model_name, train_loader, criterion, optimizer, scheduler, conn,
                device=DEVICE, epochs=NUM_EPOCHS, patience=EARLY_STOPPING_PATIENCE, target_class=None):
    print(f"\n{'='*20} Starting {model_name} Training {'='*20}")
    scaler = GradScaler() if USE_MIXED_PRECISION else None
    train_losses, train_accs, train_f1s = [], [], []
    best_train_f1 = 0
    patience_counter = 0
    all_features = None
    all_labels = None
    start_time = time.time()

    model_viz_dir = os.path.join(PLOT_DIR, model_name)
    os.makedirs(model_viz_dir, exist_ok=True)

    for epoch in range(epochs):
        print(f"\nEpoch {epoch+1}/{epochs}")
        model.train()
        running_loss = 0.0
        epoch_features = []
        epoch_labels = []
        all_preds = []
        all_true = []

        train_pbar = tqdm(train_loader, desc=f"Training {model_name} (Epoch {epoch+1}/{epochs})")
        for inputs, labels in train_pbar:
            inputs, labels = inputs.to(device), labels.to(device).float()
            optimizer.zero_grad()

            if scaler:
                with autocast(device_type=device.type):
                    logits, features = model(inputs)
                    if target_class:
                        loss = criterion(logits.squeeze(), labels)
                    else:
                        loss = criterion(logits, labels)
                scaler.scale(loss).backward()
                scaler.step(optimizer)
                scaler.update()
            else:
                logits, features = model(inputs)
                if target_class:
                    loss = criterion(logits.squeeze(), labels)
                else:
                    loss = criterion(logits, labels)
                loss.backward()
                optimizer.step()

            running_loss += loss.item()
            preds = (torch.sigmoid(logits if not target_class else logits.squeeze()) > 0.5).int()
            all_preds.append(preds.cpu().numpy())
            all_true.append(labels.cpu().numpy())

            epoch_features.append(features.detach().cpu().numpy())
            epoch_labels.append(labels.cpu().numpy())

            train_pbar.set_postfix({'loss': f"{loss.item():.4f}"})

        train_loss = running_loss / len(train_loader)
        all_preds = np.concatenate(all_preds, axis=0).astype(np.int32)
        all_true = np.concatenate(all_true, axis=0).astype(np.int32)
        correct = np.mean(all_preds == all_true) if target_class else np.all(all_preds == all_true, axis=1).mean()
        train_f1 = f1_score(all_true, all_preds, average='binary' if target_class else 'samples')

        train_losses.append(train_loss)
        train_accs.append(correct)
        train_f1s.append(train_f1)

        cm_per_class = []
        if not target_class:
            for i in range(all_true.shape[1]):
                true_1d = all_true[:, i]
                pred_1d = all_preds[:, i]
                cm = cm_cuml(cp.array(true_1d), cp.array(pred_1d))
                cm_per_class.append(cp.asnumpy(cm))
            cm = np.sum(cm_per_class, axis=0) if cm_per_class else np.zeros((2, 2))
            class_acc = {cls: float(cp.sum((cp.array(all_preds)[:, i] == cp.array(all_true)[:, i])).item() / cp.sum(cp.array(all_true)[:, i]).item()) if cp.sum(cp.array(all_true)[:, i]) > 0 else 0.0 for i, cls in enumerate(PPE_ITEMS)}
            class_f1 = {cls: f1_score(all_true[:, i], all_preds[:, i], average='binary') for i, cls in enumerate(PPE_ITEMS)}
        else:
            cm = cm_cuml(cp.array(all_true), cp.array(all_preds))
            cm = cp.asnumpy(cm)
            class_acc = {target_class: correct}
            class_f1 = {target_class: train_f1}

        print(f"Epoch {epoch+1}/{epochs}:")
        print(f"  Train - Loss: {train_loss:.4f}, Accuracy: {correct:.4f}, F1: {train_f1:.4f}")

        log_training_metrics(conn, model_name, epoch+1, train_loss, correct, train_f1, class_acc)

        if train_f1 > best_train_f1:
            best_train_f1 = train_f1
            patience_counter = 0
            torch.save({
                'epoch': epoch,
                'model_state_dict': model.state_dict(),
                'optimizer_state_dict': optimizer.state_dict(),
                'train_f1': train_f1,
                'class_accuracies': class_acc,
                'class_f1_scores': class_f1
            }, os.path.join(MODEL_DIR, f"{model_name}_best.pt"))
            print(f"  → New best model saved with F1 score: {train_f1:.4f}")
            all_features = np.concatenate(epoch_features, axis=0)
            all_labels = np.concatenate(epoch_labels, axis=0)
        else:
            patience_counter += 1

        if patience_counter >= patience:
            print(f"Early stopping at epoch {epoch+1}")
            break

        scheduler.step(train_f1)

    if all_features is not None and all_labels is not None and RUN_T_SNE:
        print(f"  Computing final t-SNE projection...")
        torch.cuda.empty_cache()
        sample_size = min(50, len(all_features))  # Reduced to 50
        indices = np.random.choice(len(all_features), sample_size, replace=False)
        features = all_features[indices]
        labels = all_labels[indices]

        # Validate input
        if np.any(np.isnan(features)) or np.any(np.isinf(features)):
            print("  Warning: NaN or Inf values detected in features. Skipping t-SNE.")
        else:
            try:
                pca = PCA(n_components=30)
                features_reduced = pca.fit_transform(features)
                tsne = TSNE(n_components=2, perplexity=15, n_iter=300, random_state=42, n_jobs=1)
                tsne_result = tsne.fit_transform(features_reduced)
                tsne_x = tsne_result[:, 0]
                tsne_y = tsne_result[:, 1]
                labels_np = labels if not target_class else labels.reshape(-1, 1)

                plt.figure(figsize=(12, 10))
                if not target_class:
                    for i, label_name in enumerate(PPE_ITEMS):
                        mask = labels_np[:, i] == 1
                        if np.any(mask):

                plt.figure(figsize=(12, 10))
                sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
                           xticklabels=['Negative', 'Positive'], yticklabels=['Negative', 'Positive'])
                plt.title(f'{model_name} - Final Confusion Matrix')
                plt.xlabel('Predicted')
                plt.ylabel('True')
                plt.tight_layout()
                plt.savefig(os.path.join(model_viz_dir, 'final_confusion.png'))
                plt.close()

                plt.figure(figsize=(12, 6))
                plt.bar(class_f1.keys(), class_f1.values(), color=CLASS_COLORS[:len(class_f1)])
                plt.title(f'{model_name} - Final Per-Class F1 Scores')
                plt.xlabel('Class')
                plt.ylabel('F1 Score')
                plt.xticks(rotation=45)
                plt.ylim(0, 1.0)
                plt.tight_layout()
                plt.savefig(os.path.join(model_viz_dir, 'final_class_f1.png'))
                plt.close()
            except Exception as e:
                print(f"  Error during t-SNE visualization: {e} - Skipping t-SNE.")

    checkpoint = torch.load(os.path.join(MODEL_DIR, f"{model_name}_best.pt"))
    model.load_state_dict(checkpoint['model_state_dict'])
    training_time = time.time() - start_time
    print(f"\n{model_name} training completed in {training_time:.2f} seconds")

    plt.figure(figsize=(15, 5))

def process_realtime(model, target_class, video_path=None, fps=30):
    cap = cv2.VideoCapture(video_path if video_path else 0)
    if not cap.isOpened():
        print("Error: Could not open video or webcam.")
        return

    transform = CUDATransform(transforms.Compose([
        transforms.ToPILImage(),
        transforms.Resize((224, 224)),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
    ]))

    log_file = os.path.join(LOG_DIR, f"{os.path.basename(video_path) if video_path else 'webcam'}_{target_class}_log.json")
    flag_segments = []
    start_time = None
    frame_count = 0

    while cap.isOpened():
        ret, frame = cap.read()
        if not ret:
            break
        frame_count += 1
        input_tensor = transform(frame).unsqueeze(0).to(DEVICE)
        with torch.no_grad():
            logits, _ = model(input_tensor)
            pred = (torch.sigmoid(logits.squeeze()) > 0.5).int().item()

        current_time = frame_count / fps
        if pred == 0:  # Target class absent
            if start_time is None:
                start_time = current_time
        else:  # Target class present
            if start_time is not None:
                end_time = current_time
                flag_segments.append({
                    "start": f"{int(start_time//60):02d}:{int(start_time%60):02d}.{int((start_time%1)*1000):03d}",
                    "end": f"{int(end_time//60):02d}:{int(end_time%60):02d}.{int((end_time%1)*1000):03d}",
                    "missing": target_class
                })
                start_time = None

    if start_time is not None:
        end_time = frame_count / fps
        flag_segments.append({
            "start": f"{int(start_time//60):02d}:{int(start_time%60):02d}.{int((start_time%1)*1000):03d}",
            "end": f"{int(end_time//60):02d}:{int(end_time%60):02d}.{int((end_time%1)*1000):03d}",
            "missing": target_class
        })

    status = "accepted" if not flag_segments else "flagged"
    log_data = {
        "source": os.path.basename(video_path) if video_path else "webcam",
        "target_class": target_class,
        "status": status,
        "flag_segments": flag_segments,
        "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "frame_rate": fps
    }

    with open(log_file, 'w') as f:
        json.dump(log_data, f, indent=4)
    print(f"Processed {video_path or 'webcam'} for {target_class}. Status: {status}. Log saved to {log_file}")

    cap.release()

def visualize_combined_features(results_dict, save_dir=PLOT_DIR):
    model_names = list(results_dict.keys())
    fig, axes = plt.subplots(1, len(model_names), figsize=(20, 7))
    if len(model_names) == 1:
        axes = [axes]

    for i, name in enumerate(model_names):
        model_features = results_dict[name]['features']
        model_labels = results_dict[name]['labels']

        if len(model_features) > 0 and RUN_T_SNE:
            sample_size = min(50, len(model_features))
            indices = np.random.choice(len(model_features), sample_size, replace=False)
            features = model_features[indices]
            labels = model_labels[indices]

            if np.any(np.isnan(features)) or np.any(np.isinf(features)):
                print(f"  Warning: NaN or Inf values in {name} features. Skipping visualization.")
            else:
                try:
                    pca = PCA(n_components=30)
                    features_reduced = pca.fit_transform(features)
                    tsne = TSNE(n_components=2, perplexity=15, n_iter=300, random_state=42, n_jobs=1)
                    tsne_result = tsne.fit_transform(features_reduced)
                    tsne_x = tsne_result[:, 0]
                    tsne_y = tsne_result[:, 1]

                    for j, label_name in enumerate(PPE_ITEMS):
                        mask = labels[:, j] == 1
                        if np.any(mask):
                            axes[i].scatter(tsne_x[mask], tsne_y[mask],
                                         label=label_name, color=CLASS_COLORS[j], alpha=0.6)

                    axes[i].set_title(f'{name}\nFeature Space Visualization')
                    axes[i].set_xlabel('t-SNE Component 1')
                    axes[i].set_ylabel('t-SNE Component 2')
                    if i == len(model_names) - 1:
                        axes[i].legend(bbox_to_anchor=(1.05, 1), loc='upper left')
                except Exception as e:
                    print(f"  Error during t-SNE visualization for {name}: {e}")
                    axes[i].text(0.5, 0.5, f"Visualization failed: {str(e)}",
                              ha='center', va='center', transform=axes[i].transAxes)

    plt.tight_layout()
    plt.savefig(os.path.join(save_dir, 'combined_tsne_visualization.png'))
    plt.close()

    print("Skipping 3D t-SNE and cluster analysis to conserve memory.")

def main():
    parser = argparse.ArgumentParser(description="RAPIDS-accelerated PPE Detection Training and Real-time Processing")
    parser.add_argument("--target-class", type=str, choices=PPE_ITEMS, help="Specify target class for single-object detection")
    parser.add_argument("--video", type=str, help="Path to video file for real-time processing (optional)")
    args = parser.parse_args()

    print("\n" + "="*50)
    print("RAPIDS-accelerated PPE Detection Training and Real-time Processing")
    print("="*50 + "\n")

    start_time = time.time()

    conn = connect_db()

    try:
        print(f"\nLoading COCO dataset from {TRAIN_ANN_FILE}...")
        coco = COCO(TRAIN_ANN_FILE)
        if not coco.dataset or not coco.getImgIds():
            raise ValueError(f"No images or annotations found in {TRAIN_ANN_FILE}. Verify the file exists and contains valid COCO data.")

        data_dir = [TRAIN_DATA_DIR]

        print(f"Categories: {len(coco.cats)}")
        print(f"Images: {len(coco.imgs)}")
        print(f"Annotations: {len(coco.anns)}")

        class_dist = {cat['name']: len(coco.getAnnIds(catIds=cat_id))
                     for cat_id, cat in coco.cats.items()}
        print("Class distribution:")
        for cls, count in class_dist.items():
            print(f"  {cls}: {count}")

        plt.figure(figsize=(12, 8))
        plt.bar(class_dist.keys(), class_dist.values(), color=CLASS_COLORS)
        plt.title('Class Distribution in Train Dataset')
        plt.xlabel('Class')
        plt.ylabel('Count')
        plt.xticks(rotation=45)
        plt.tight_layout()
        plt.savefig(os.path.join(PLOT_DIR, 'class_distribution.png'))
        plt.close()

        print("\nCreating dataset and dataloader...")
        train_dataset = CocoDataset(coco, TRAIN_DATA_DIR, transform=cuda_transform, phase='train', target_class=args.target_class)
        if len(train_dataset) == 0:
            raise ValueError(f"Dataset is empty. Verify images exist in {TRAIN_DATA_DIR} and match {TRAIN_ANN_FILE}.")

        train_size = len(train_dataset)
        train_sampler = BalancedSampler(train_dataset, oversample_factor=2.0) if not args.target_class else None
        train_loader = DataLoader(train_dataset, batch_size=BATCH_SIZE, sampler=train_sampler, shuffle=not train_sampler, num_workers=2, pin_memory=False, prefetch_factor=1)

        print(f"Train dataset size: {train_size}")

        print("\nInitializing models...")
        models = {
            'CustomCNN': EnhancedCustomCNN(num_classes=1 if args.target_class else len(PPE_ITEMS)).to(DEVICE),
            'ResNet18': ModifiedResNet(num_classes=1 if args.target_class else len(PPE_ITEMS)).to(DEVICE),
            'EfficientNet': EfficientNetModel(num_classes=1 if args.target_class else len(PPE_ITEMS)).to(DEVICE)
        }

        results = {}
        for name, model in models.items():
            print(f"\nPreparing to train {name}...")
            criterion = nn.BCEWithLogitsLoss()
            optimizer = torch.optim.AdamW(model.parameters(), lr=LEARNING_RATE, weight_decay=WEIGHT_DECAY)
            scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(optimizer, mode='max', factor=0.5, patience=2, verbose=True)

            results[name] = train_model(model, name, train_loader, criterion, optimizer, scheduler, conn, target_class=args.target_class)

        print("\nGenerating comparative visualizations...")
        plot_training_metrics(results)
        visualize_combined_features(results)

        summary = {
            'timestamp': datetime.now().strftime("%Y-%m-%d %H:M:%S"),
            'total_runtime': time.time() - start_time,
            'models': list(models.keys()),
            'training_results': {name: {'best_f1': results[name]['best_train_f1'], 'training_time': results[name]['training_time']} for name in models.keys()}
        }

        with open(os.path.join(OUTPUT_DIR, 'training_summary.json'), 'w') as f:
            json.dump(summary, f, indent=4)

        if args.video:
            print(f"\nProcessing video: {args.video}...")
            for name, model in models.items():
                if args.target_class:
                    checkpoint = torch.load(os.path.join(MODEL_DIR, f"{name}_best.pt"))
                    model.load_state_dict(checkpoint['model_state_dict'])
                    process_realtime(model, args.target_class, args.video)

        print("\n" + "="*50)
        print(f"Completed in {time.time() - start_time:.2f} seconds")
        print(f"All results saved to {OUTPUT_DIR}")
        print("="*50 + "\n")

    except Exception as e:
        print(f"\nError during execution: {e}")
        import traceback
        traceback.print_exc()
    finally:
        if conn:
            conn.close()
            print("Database connection closed.")

if __name__ == "__main__":
    main()

async def process_frames(sid):
    while state["is_running"] and state["cap"]:
        ret, frame = state["cap"].read()
        if not ret:
            logger.error("Failed to capture frame")
            await sio.emit("error", {"message": "Failed to capture frame"}, to=sid)
            break
        frame, stats, status = process_frame(frame)
        _, buffer = cv2.imencode(".jpg", frame)
        frame_b64 = base64.b64encode(buffer).decode("utf-8")
        session_duration = time.time() - state["start_time"]
        hours, remainder = divmod(int(session_duration), 3600)
        minutes, seconds = divmod(remainder, 60)
        session_time = f"{hours:02d}:{minutes:02d}:{seconds:02d}"
        await sio.emit("frame", {
            "frame": frame_b64,
            "stats": stats,
            "sessionTime": session_time,
            "status": status.capitalize()
        }, to=sid)
        state["frame_count"] += 1
        await asyncio.sleep(0.033)

def process_frame(frame):
    face_rois, face_coords = detect_faces(frame)
    stats = state["ppe_stats"]
    status = "No status detected yet"
    for face_roi, (x, y, w, h) in zip(face_rois, face_coords):
        pil_img = Image.fromarray(cv2.cvtColor(face_roi, cv2.COLOR_BGR2RGB))
        img_tensor = transform(pil_img).unsqueeze(0).to(device)
        with torch.no_grad():
            logits, _ = state["model"](img_tensor)
            if state["num_classes"] > 1:
                probs = torch.sigmoid(logits).squeeze()
                detected_items = [PPE_ITEMS[i] for i in range(len(probs)) if probs[i] >= DETECTION_CONFIDENCE_THRESHOLD]
                confidence = max(probs.tolist()) if detected_items else 0.0
            else:
                prob = torch.sigmoid(logits.squeeze()).item()
                detected = 1 if prob >= DETECTION_CONFIDENCE_THRESHOLD else 0
                detected_items = [state["target_class"]] if detected else []
                confidence = prob
        if detected_items:
            state["ppe_history"].add_ppe(detected_items, confidence)
            status, acceptance_rate, flagged_items = state["ppe_history"].get_status(state["target_class"], state["num_classes"])
            if status == "Accepted":
                stats["Accepted"] += 1
                color = PPE_COLORS[state["target_class"]] if state["num_classes"] == 1 else (0, 255, 0)
            else:
                stats["Flagged"] += 1
                color = (0, 0, 255)
            cv2.rectangle(frame, (x, y), (x+w, y+h), color, 2)
            status_text = f"{status}: {acceptance_rate:.1f}%"
            if flagged_items:
                status_text += f" | Missing: {', '.join(flagged_items)}"
            cv2.putText(frame, status_text, (x, y-10), cv2.FONT_HERSHEY_SIMPLEX, 0.7, color, 2)
    return frame, stats, status

def record_session_data():
    if not state["start_time"] or state["frame_count"] == 0:
        logger.info("No session data to record: start_time or frame_count is invalid")
        return
    try:

# PPE History class
class PPEHistory:
    def __init__(self, history_size=5):
        self.history_size = history_size
        self.ppe_history = []

    def add_ppe(self, detected_items, confidence):
        self.ppe_history.append((detected_items, confidence))
        if len(self.ppe_history) > self.history_size:
            self.ppe_history.pop(0)

    def get_status(self, target_class, num_classes=1):
        if not self.ppe_history:
            return None, 0.0, []
        accepted_count = 0
        flagged_items = []
        for detected_items, confidence in self.ppe_history:
            if num_classes > 1:
                if target_class in detected_items:
                    accepted_count += 1
                else:
                    flagged_items.append(target_class)
            else:
                if confidence >= DETECTION_CONFIDENCE_THRESHOLD:
                    accepted_count += 1
                else:
                    flagged_items.append(target_class)
        total = len(self.ppe_history)
        acceptance_rate = (accepted_count / total) * 100 if total > 0 else 0.0
        return "Accepted" if acceptance_rate >= 50 else "Flagged", acceptance_rate, list(set(flagged_items))
